# Dependencias do agente de impressao. Deliberadamente curtas: este pacote
# e instalado a mao no computador do balcao, muitas vezes por telefone, e
# cada dependencia a mais e uma chance de a instalacao falhar la.
#
# NAO tem nada em comum com o requirements.txt da API. O agente nao importa
# FastAPI, SQLAlchemy nem nada do backend — ele fala com a API por HTTP.

requests>=2.31

# Acesso a fila de impressao do Windows (win32print). So instala no Windows;
# em Linux/macOS o pip pula esta linha, e o agente ainda roda com --dry-run
# para quem esta desenvolvendo.
pywin32>=306; sys_platform == "win32"

# Icone na bandeja, ao lado do relogio. Substituiu a janela preta do console
# (ver build.bat). O Pillow vem junto porque o pystray recebe a imagem do
# icone como objeto PIL — nao ha como usar um sem o outro.
#
# So no Windows pelo mesmo motivo do pywin32: no Linux o pystray exige um
# servidor grafico, e quem desenvolve fora do Windows roda com --no-tray. O
# `__main__` cai sozinho nesse modo se o import falhar.
pystray>=0.19; sys_platform == "win32"
pillow>=10.0; sys_platform == "win32"
